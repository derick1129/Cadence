**Goal:** Implement database-backed agent configuration with immutable versioning, catalog validation, RBAC, and strict cross-tenant isolation.

**Architecture:** Domain-driven modular monolith following layered clean architecture (`router` -> `service` -> `repository`). SQLAlchemy 2.0 async declarative models (`Agent`, `AgentVersion`) backed by PostgreSQL, Alembic migrations, immutable version bumping on prompt or configuration change, and tenant validation via JWT claims (`get_current_tenant_id`) and RBAC (`require_roles`).

- Config comes only from the settings module (`pydantic-settings`). No hardcoded secrets or URLs.
- Every tenant-scoped query must be filtered by `organization_id` (UUID, non-null, indexed).

- Roles: `owner` (billing, delete org), `admin` (manage agents, members), `member` (use and view).
- Agent versions are **immutable**: once created, an `AgentVersion` record is never mutated; updates create a new version with an incremented `version_no`.

```python
"""Agent and AgentVersion SQLAlchemy models."""

import uuid
from typing import Any

from sqlalchemy import ForeignKey, Index, Integer, String, Text, UniqueConstraint
from sqlalchemy.dialects.postgresql import JSONB, UUID
from sqlalchemy.orm import Mapped, mapped_column

from backend.app.core.base import Base, TimestampMixin


class Agent(Base, TimestampMixin):
    """Voice agent entity scoped strictly to an organization tenant."""

    __tablename__ = "agents"
    __table_args__ = (
        Index("ix_agents_organization_id", "organization_id"),
    )

    organization_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("organizations.id", ondelete="CASCADE"),
        nullable=False,
    )
    name: Mapped[str] = mapped_column(String(255), nullable=False)
    current_version_id: Mapped[uuid.UUID | None] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("agent_versions.id", ondelete="SET NULL", use_alter=True, name="fk_agents_current_version"),
        nullable=True,
    )


class AgentVersion(Base, TimestampMixin):
    """Immutable version configuration of an agent."""

    __tablename__ = "agent_versions"
    __table_args__ = (
        UniqueConstraint("agent_id", "version_no", name="uq_agent_version_no"),
        Index("ix_agent_versions_agent_id", "agent_id"),
    )

    agent_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("agents.id", ondelete="CASCADE"),
        nullable=False,
    )
    version_no: Mapped[int] = mapped_column(Integer, nullable=False)
    system_prompt: Mapped[str] = mapped_column(Text, nullable=False)
    greeting: Mapped[str | None] = mapped_column(Text, nullable=True)
    stt_config: Mapped[dict[str, Any]] = mapped_column(JSONB, nullable=False, default=dict)
    llm_config: Mapped[dict[str, Any]] = mapped_column(JSONB, nullable=False, default=dict)
    tts_config: Mapped[dict[str, Any]] = mapped_column(JSONB, nullable=False, default=dict)
    silence_timeout_s: Mapped[int] = mapped_column(Integer, default=15, nullable=False)
    max_duration_s: Mapped[int] = mapped_column(Integer, default=600, nullable=False)
    created_by: Mapped[uuid.UUID | None] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("users.id", ondelete="SET NULL"),
        nullable=True,
    )
```

```python
"""Agent repository handling persistence for agents and immutable versions."""

import uuid
from datetime import UTC, datetime
from typing import Any

from sqlalchemy import desc, select, update
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.modules.agents.models import Agent, AgentVersion


class AgentRepository:
    """Repository handling persistence for agents and immutable versions."""

    def __init__(self, session: AsyncSession | None = None) -> None:
        self.session = session
        self._agents: dict[str, dict[str, Any]] = {}
        self._versions: dict[str, dict[str, Any]] = {}

    async def list_agents(self, organization_id: str) -> list[dict[str, Any]]:
        """List all agents belonging to an organization."""
        if self.session is not None:
            try:
                org_uuid = uuid.UUID(organization_id)
            except ValueError:
                return []
            result = await self.session.execute(
                select(Agent)
                .where(Agent.organization_id == org_uuid)
                .order_by(desc(Agent.created_at))
            )
            records = result.scalars().all()
            return [
                {
                    "id": str(r.id),
                    "organization_id": str(r.organization_id),
                    "name": r.name,
                    "current_version_id": str(r.current_version_id) if r.current_version_id else None,
                    "created_at": r.created_at,
                    "updated_at": r.updated_at,
                }
                for r in records
            ]
        return [
            agent
            for agent in self._agents.values()
            if agent.get("organization_id") == organization_id
        ]

    async def get_agent_by_id(self, organization_id: str, agent_id: str) -> dict[str, Any] | None:
        """Fetch an agent by ID scoped strictly to the organization."""
        if self.session is not None:
            try:
                org_uuid = uuid.UUID(organization_id)
                agent_uuid = uuid.UUID(agent_id)
            except ValueError:
                return None
            result = await self.session.execute(
                select(Agent).where(
                    Agent.id == agent_uuid,
                    Agent.organization_id == org_uuid,
                )
            )
            record = result.scalar_one_or_none()
            if not record:
                return None
            return {
                "id": str(record.id),
                "organization_id": str(record.organization_id),
                "name": record.name,
                "current_version_id": str(record.current_version_id) if record.current_version_id else None,
                "created_at": record.created_at,
                "updated_at": record.updated_at,
            }
        agent = self._agents.get(agent_id)
        if not agent or agent.get("organization_id") != organization_id:
            return None
        return agent

    async def create_agent(self, organization_id: str, name: str) -> dict[str, Any]:
        """Create a new agent row for the specified tenant."""
        now = datetime.now(UTC)
        if self.session is not None:
            org_uuid = uuid.UUID(organization_id)
            agent = Agent(organization_id=org_uuid, name=name)
            self.session.add(agent)
            await self.session.flush()
            return {
                "id": str(agent.id),
                "organization_id": str(agent.organization_id),
                "name": agent.name,
                "current_version_id": None,
                "created_at": agent.created_at,
                "updated_at": agent.updated_at,
            }
        agent_id = str(uuid.uuid4())
        agent_dict = {
            "id": agent_id,
            "organization_id": organization_id,
            "name": name,
            "current_version_id": None,
            "created_at": now,
            "updated_at": now,
        }
        self._agents[agent_id] = agent_dict
        return agent_dict

    async def set_current_version(
        self, organization_id: str, agent_id: str, version_id: str
    ) -> dict[str, Any] | None:
        """Update current_version_id on an existing agent."""
        if self.session is not None:
            try:
                org_uuid = uuid.UUID(organization_id)
                agent_uuid = uuid.UUID(agent_id)
                ver_uuid = uuid.UUID(version_id)
            except ValueError:
                return None
            await self.session.execute(
                update(Agent)
                .where(Agent.id == agent_uuid, Agent.organization_id == org_uuid)
                .values(current_version_id=ver_uuid, updated_at=datetime.now(UTC))
            )
            await self.session.flush()
            return await self.get_agent_by_id(organization_id, agent_id)

        agent = await self.get_agent_by_id(organization_id, agent_id)
        if not agent:
            return None
        agent["current_version_id"] = version_id
        agent["updated_at"] = datetime.now(UTC)
        return agent

    async def update_agent_name(
        self, organization_id: str, agent_id: str, name: str
    ) -> dict[str, Any] | None:
        """Update agent name."""
        if self.session is not None:
            try:
                org_uuid = uuid.UUID(organization_id)
                agent_uuid = uuid.UUID(agent_id)
            except ValueError:
                return None
            await self.session.execute(
                update(Agent)
                .where(Agent.id == agent_uuid, Agent.organization_id == org_uuid)
                .values(name=name, updated_at=datetime.now(UTC))
            )
            await self.session.flush()
            return await self.get_agent_by_id(organization_id, agent_id)

        agent = await self.get_agent_by_id(organization_id, agent_id)
        if not agent:
            return None
        agent["name"] = name
        agent["updated_at"] = datetime.now(UTC)
        return agent

        self,
        agent_id: str,
        version_no: int,
        system_prompt: str,
        greeting: str | None,
        stt_config: dict[str, Any],
        llm_config: dict[str, Any],
        tts_config: dict[str, Any],
        silence_timeout_s: int = 15,
        max_duration_s: int = 600,
        created_by: str | None = None,
    ) -> dict[str, Any]:
        """Create an immutable version row for an agent."""
        now = datetime.now(UTC)
        if self.session is not None:
            agent_uuid = uuid.UUID(agent_id)
            user_uuid = uuid.UUID(created_by) if created_by else None
            version = AgentVersion(
                agent_id=agent_uuid,
                version_no=version_no,
                system_prompt=system_prompt,
                greeting=greeting,
                stt_config=stt_config,
                llm_config=llm_config,
                tts_config=tts_config,
                silence_timeout_s=silence_timeout_s,
                max_duration_s=max_duration_s,
                created_by=user_uuid,
            )
            self.session.add(version)
            await self.session.flush()
            return {
                "id": str(version.id),
                "agent_id": str(version.agent_id),
                "version_no": version.version_no,
                "system_prompt": version.system_prompt,
                "greeting": version.greeting,
                "stt_config": version.stt_config,
                "llm_config": version.llm_config,
                "tts_config": version.tts_config,
                "silence_timeout_s": version.silence_timeout_s,
                "max_duration_s": version.max_duration_s,
                "created_by": str(version.created_by) if version.created_by else None,
                "created_at": version.created_at,
            }

        version_id = str(uuid.uuid4())
        ver_dict = {
            "id": version_id,
            "agent_id": agent_id,
            "version_no": version_no,
            "system_prompt": system_prompt,
            "greeting": greeting,
            "stt_config": stt_config,
            "llm_config": llm_config,
            "tts_config": tts_config,
            "silence_timeout_s": silence_timeout_s,
            "max_duration_s": max_duration_s,
            "created_by": created_by,
            "created_at": now,
        }
        self._versions[version_id] = ver_dict
        return ver_dict

    async def get_version_by_id(self, version_id: str) -> dict[str, Any] | None:
        """Fetch a version record by its ID."""
        if self.session is not None:
            try:
                ver_uuid = uuid.UUID(version_id)
            except ValueError:
                return None
            result = await self.session.execute(

            )
            v = result.scalar_one_or_none()
            if not v:
                return None
            return {
                "id": str(v.id),
                "agent_id": str(v.agent_id),
                "version_no": v.version_no,
                "system_prompt": v.system_prompt,
                "greeting": v.greeting,
                "stt_config": v.stt_config,
                "llm_config": v.llm_config,
                "tts_config": v.tts_config,
                "silence_timeout_s": v.silence_timeout_s,
                "max_duration_s": v.max_duration_s,
                "created_by": str(v.created_by) if v.created_by else None,
                "created_at": v.created_at,
            }
        return self._versions.get(version_id)

                    "stt_config": v.stt_config,

                    "silence_timeout_s": v.silence_timeout_s,
                    "max_duration_s": v.max_duration_s,
                    "created_by": str(v.created_by) if v.created_by else None,
                    "created_at": v.created_at,
                }
                for v in records
            ]
        versions = [v for v in self._versions.values() if v.get("agent_id") == agent_id]
        versions.sort(key=lambda x: int(x.get("version_no", 0)))
        return versions

    async def get_latest_version_no(self, agent_id: str) -> int:
        """Get the highest version number for an agent."""
        if self.session is not None:
            try:
                agent_uuid = uuid.UUID(agent_id)
            except ValueError:
                return 0
            result = await self.session.execute(
                select(AgentVersion.version_no)
                .where(AgentVersion.agent_id == agent_uuid)
                .order_by(desc(AgentVersion.version_no))
                .limit(1)
            )
            val = result.scalar_one_or_none()
            return val if val is not None else 0
        versions = await self.list_versions_by_agent_id(agent_id)
        if not versions:
            return 0
        return max(int(v.get("version_no", 0)) for v in versions)
```

```python
"""Agent business logic service."""

from typing import Any

from backend.app.core.errors import AppError
from backend.app.modules.agents.repository import AgentRepository
from backend.app.modules.agents.schemas import (
    AgentCreate,
    AgentResponse,
    AgentUpdate,
    AgentVersionResponse,
)
from backend.app.modules.providers.service import ProviderService


DEFAULT_STT_CONFIG = {"provider": "deepgram", "model": "nova-2"}
DEFAULT_LLM_CONFIG = {"provider": "openai", "model": "gpt-4o-mini"}
DEFAULT_TTS_CONFIG = {"provider": "cartesia", "model": "sonic-english", "voice": "default-voice"}


class AgentService:
    """Service handling business rules, version bumping, and agent queries."""

    def __init__(
        self,
        repository: AgentRepository | None = None,
        provider_service: ProviderService | None = None,
    ) -> None:
        self.repository = repository or AgentRepository()
        self.provider_service = provider_service or ProviderService()

    async def _validate_configs(
        self,
        stt_config: dict[str, Any],
        llm_config: dict[str, Any],
        tts_config: dict[str, Any],
    ) -> tuple[dict[str, Any], dict[str, Any], dict[str, Any]]:
        """Validate or populate default STT, LLM, and TTS configurations against the provider catalog."""
        stt = {**DEFAULT_STT_CONFIG, **stt_config}
        llm = {**DEFAULT_LLM_CONFIG, **llm_config}
        tts = {**DEFAULT_TTS_CONFIG, **tts_config}

        catalog = await self.provider_service.get_catalog()

        stt_models = {m for p in catalog.stt for m in p.models}
        if stt.get("model") not in stt_models:
            raise AppError(
                code="INVALID_PROVIDER_CONFIG",
                message=f"STT model '{stt.get('model')}' is not in catalog. Supported: {sorted(stt_models)}",
                status_code=400,
            )

        llm_models = {m for p in catalog.llm for m in p.models}
        if llm.get("model") not in llm_models:
            raise AppError(
                code="INVALID_PROVIDER_CONFIG",
                message=f"LLM model '{llm.get('model')}' is not in catalog. Supported: {sorted(llm_models)}",
                status_code=400,
            )

        tts_models = {m for p in catalog.tts for m in p.models}
        if tts.get("model") not in tts_models:
            raise AppError(
                code="INVALID_PROVIDER_CONFIG",
                message=f"TTS model '{tts.get('model')}' is not in catalog. Supported: {sorted(tts_models)}",
                status_code=400,
            )

        return stt, llm, tts

    async def list_agents(self, organization_id: str) -> list[AgentResponse]:
        """List all agents for an organization with their current pinned version."""
        agents = await self.repository.list_agents(organization_id)
        results: list[AgentResponse] = []
        for agent in agents:
            current_version: AgentVersionResponse | None = None
            if agent.get("current_version_id"):
                raw_version = await self.repository.get_version_by_id(str(agent["current_version_id"]))
                if raw_version:
                    current_version = AgentVersionResponse(**raw_version)
            results.append(
                AgentResponse(
                    id=agent["id"],
                    organization_id=agent["organization_id"],
                    name=agent["name"],
                    current_version_id=agent.get("current_version_id"),
                    current_version=current_version,
                    created_at=agent["created_at"],
                    updated_at=agent["updated_at"],
                )
            )
        return results

    async def get_agent(self, organization_id: str, agent_id: str) -> AgentResponse | None:
        """Retrieve an agent by ID scoped to the tenant."""
        agent = await self.repository.get_agent_by_id(organization_id, agent_id)
        if not agent:
            return None
        current_version: AgentVersionResponse | None = None
        if agent.get("current_version_id"):
            raw_version = await self.repository.get_version_by_id(str(agent["current_version_id"]))
            if raw_version:
                current_version = AgentVersionResponse(**raw_version)
        return AgentResponse(
            id=agent["id"],
            organization_id=agent["organization_id"],
            name=agent["name"],
            current_version_id=agent.get("current_version_id"),
            current_version=current_version,
            created_at=agent["created_at"],
            updated_at=agent["updated_at"],
        )

    async def create_agent(
        self,
        organization_id: str,
        data: AgentCreate,
        user_id: str | None = None,
    ) -> AgentResponse:
        """Create a new agent along with its initial version (v1)."""
        stt, llm, tts = await self._validate_configs(data.stt_config, data.llm_config, data.tts_config)

        agent = await self.repository.create_agent(
            organization_id=organization_id,
            name=data.name,
        )
        version = await self.repository.create_version(
            agent_id=agent["id"],
            version_no=1,
            system_prompt=data.system_prompt,
            greeting=data.greeting,
            stt_config=stt,
            llm_config=llm,
            tts_config=tts,
            silence_timeout_s=data.silence_timeout_s,
            max_duration_s=data.max_duration_s,
            created_by=user_id,
        )
        updated_agent = await self.repository.set_current_version(
            organization_id=organization_id,
            agent_id=agent["id"],
            version_id=version["id"],
        )
        agent_data = updated_agent or agent
        return AgentResponse(
            id=agent_data["id"],
            organization_id=agent_data["organization_id"],
            name=agent_data["name"],
            current_version_id=version["id"],
            current_version=AgentVersionResponse(**version),
            created_at=agent_data["created_at"],
            updated_at=agent_data["updated_at"],
        )

    async def update_agent(
        self,
        organization_id: str,
        agent_id: str,
        data: AgentUpdate,
        user_id: str | None = None,
    ) -> AgentResponse | None:
        """Update agent metadata or create a new immutable version if prompt/configs changed."""
        agent = await self.repository.get_agent_by_id(organization_id, agent_id)
        if not agent:
            return None

        if data.name is not None and data.name != agent["name"]:
            await self.repository.update_agent_name(organization_id, agent_id, data.name)

        has_version_change = any(
            x is not None
            for x in (
                data.system_prompt,
                data.greeting,
                data.stt_config,
                data.llm_config,
                data.tts_config,
                data.silence_timeout_s,
                data.max_duration_s,
            )
        )

        if has_version_change:
            latest_version_dict: dict[str, Any] = {}
            if agent.get("current_version_id"):
                v = await self.repository.get_version_by_id(str(agent["current_version_id"]))
                if v:
                    latest_version_dict = v

            stt, llm, tts = await self._validate_configs(
                data.stt_config if data.stt_config is not None else latest_version_dict.get("stt_config", {}),
                data.llm_config if data.llm_config is not None else latest_version_dict.get("llm_config", {}),
                data.tts_config if data.tts_config is not None else latest_version_dict.get("tts_config", {}),
            )

            next_version_no = await self.repository.get_latest_version_no(agent_id) + 1
            new_version = await self.repository.create_version(
                agent_id=agent_id,
                version_no=next_version_no,
                system_prompt=data.system_prompt if data.system_prompt is not None else latest_version_dict.get("system_prompt", ""),
                greeting=data.greeting if data.greeting is not None else latest_version_dict.get("greeting"),
                stt_config=stt,
                llm_config=llm,
                tts_config=tts,
                silence_timeout_s=data.silence_timeout_s if data.silence_timeout_s is not None else latest_version_dict.get("silence_timeout_s", 15),
                max_duration_s=data.max_duration_s if data.max_duration_s is not None else latest_version_dict.get("max_duration_s", 600),
                created_by=user_id,
            )
            await self.repository.set_current_version(organization_id, agent_id, new_version["id"])

        return await self.get_agent(organization_id, agent_id)

    async def list_versions(self, organization_id: str, agent_id: str) -> list[AgentVersionResponse]:
        """List all versions for an agent after verifying tenant ownership."""
        agent = await self.repository.get_agent_by_id(organization_id, agent_id)
        if not agent:
            return []
        versions = await self.repository.list_versions_by_agent_id(agent_id)
        return [AgentVersionResponse(**v) for v in versions]

    async def get_version(self, organization_id: str, agent_id: str, version_id: str) -> AgentVersionResponse | None:
        """Retrieve a specific version of an agent ensuring tenant ownership."""
        agent = await self.repository.get_agent_by_id(organization_id, agent_id)
        if not agent:
            return None
        version = await self.repository.get_version_by_id(version_id)
        if not version or version.get("agent_id") != agent_id:
            return None
        return AgentVersionResponse(**version)
```
